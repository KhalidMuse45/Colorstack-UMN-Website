import React from 'react';

/** Staggers its children in when the group scrolls into view. */
export function AnimatedGroup({
  preset = 'blur-slide', stagger = 0.05, delay = 0, once = true,
  minColumnWidth, gap = 20, as: Tag = 'div', children, style, ...rest
}) {
  const n = (v, d) => (v === undefined || v === null || v === '' ? d : Number(v));
  const stg = n(stagger, 0.05), dly = n(delay, 0);
  const grid = minColumnWidth
    ? { display: 'grid', gridTemplateColumns: `repeat(auto-fit,minmax(${n(minColumnWidth, 240)}px,1fr))`, gap: n(gap, 20) }
    : null;
  const ref = React.useRef(null);
  const [on, setOn] = React.useState(false);

  React.useEffect(() => {
    const el = ref.current;
    if (!el) return;
    if (typeof IntersectionObserver === 'undefined' || reduced()) { setOn(true); return; }
    const io = new IntersectionObserver(([e]) => {
      if (e.isIntersecting) { setOn(true); if (once) io.disconnect(); }
      else if (!once) setOn(false);
    }, { threshold: 0.06 });
    io.observe(el);
    const failsafe = setTimeout(() => setOn(true), 2500);
    return () => { clearTimeout(failsafe); io.disconnect(); };
  }, [once]);

  const hidden = {
    'blur-slide': { opacity: 0, filter: 'blur(12px)', transform: 'translateY(-46px) rotateX(52deg)' },
    rise: { opacity: 0, transform: 'translateY(28px)' },
    scale: { opacity: 0, transform: 'scale(0.86)' },
    fade: { opacity: 0 },
  }[preset] || { opacity: 0 };

  const items = React.Children.toArray(children).filter(c => c !== null && c !== undefined && !(typeof c === 'string' && c.trim() === ''));
  return (
    <Tag ref={ref} style={{ perspective: '1000px', ...grid, ...style }} {...rest}>
      {items.map((child, i) => (
        <div
          key={i}
          style={{
            transformStyle: 'preserve-3d', willChange: 'opacity, transform, filter',
            transition: reduced() ? 'none' : 'opacity 900ms cubic-bezier(.22,1,.36,1), transform 900ms cubic-bezier(.34,1.3,.5,1), filter 700ms cubic-bezier(.22,1,.36,1)',
            transitionDelay: `${dly + i * stg}s`,
            ...(on ? { opacity: 1, filter: 'blur(0px)', transform: 'none' } : hidden),
          }}
        >{child}</div>
      ))}
    </Tag>
  );
}

function reduced() {
  return typeof window !== 'undefined' && window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}
