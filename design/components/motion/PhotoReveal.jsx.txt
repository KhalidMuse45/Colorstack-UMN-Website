import React from 'react';
import { ProgressiveBlur } from './ProgressiveBlur.jsx';

/**
 * Photo tile whose caption sits behind a progressive blur and rises on hover.
 * Children are the caption content.
 */
export function PhotoReveal({
  src, alt = '', height = 380, width, radius = 14, blurIntensity = 6, blurHeight = '50%', objectPosition = 'center',
  always = false, filter = 'saturate(1.04) contrast(1.03) brightness(1.02)',
  children, style, ...rest
}) {
  const len = v => (v === null || v === undefined || v === '' ? undefined : (/^-?[\d.]+$/.test(String(v)) ? Number(v) : v));
  const [hover, setHover] = React.useState(false);
  const [touch, setTouch] = React.useState(false);
  const showCaption = always === true || always === 'true' || hover || touch;
  const r = Number(radius) || 14;

  return (
    <figure
      onMouseEnter={() => setHover(true)}
      onMouseLeave={() => setHover(false)}
      onFocus={() => setHover(true)}
      onBlur={() => setHover(false)}
      onTouchStart={() => setTouch(t => !t)}
      tabIndex={0}
      style={{
        position: 'relative', margin: 0, overflow: 'hidden', borderRadius: r,
        height: len(height),
        ...(width ? { width: len(width) } : null),
        background: '#F3E9DA', cursor: 'default', ...style,
      }}
      {...rest}
    >
      <img
        src={src}
        alt={alt}
        style={{
          position: 'absolute', inset: 0, width: '100%', height: '100%',
          objectFit: 'cover', objectPosition, filter,
          transform: hover ? 'scale(1.045)' : 'none',
          transition: 'transform 900ms cubic-bezier(.22,1,.36,1)',
        }}
      />
      <ProgressiveBlur
        blurIntensity={blurIntensity}
        visible={true}
        style={{ bottom: 0, left: 0, width: '100%', height: blurHeight }}
      />
      <figcaption
        style={{
          position: 'absolute', left: 0, right: 0, bottom: 0, padding: '20px 22px 18px',
          opacity: showCaption ? 1 : 0,
          transform: showCaption ? 'none' : 'translateY(14px)',
          transition: 'opacity 320ms ease-out, transform 460ms cubic-bezier(.22,1,.36,1)',
        }}
      >{children}</figcaption>
    </figure>
  );
}
