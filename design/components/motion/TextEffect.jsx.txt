import React from 'react';

/** Reveals text one character or word at a time when it scrolls into view. */
export function TextEffect({
  text, per = 'char', preset = 'fade', delay = 0, stagger, once = true,
  as: Tag = 'span', children, style, ...rest
}) {
  const ref = React.useRef(null);
  const [on, setOn] = React.useState(false);
  const source = typeof text === 'string' ? text : flatten(children);
  const n = (v, d) => (v === undefined || v === null || v === '' ? d : Number(v));
  const step = n(stagger, per === 'char' ? 0.018 : 0.055);
  const delayN = n(delay, 0);

  React.useEffect(() => {
    const el = ref.current;
    if (!el) return;
    if (typeof IntersectionObserver === 'undefined' || reduced()) { setOn(true); return; }
    const io = new IntersectionObserver(([e]) => {
      if (e.isIntersecting) { setOn(true); if (once) io.disconnect(); }
      else if (!once) setOn(false);
    }, { threshold: 0.06 });
    io.observe(el);
    const failsafe = setTimeout(() => setOn(true), 2500);
    return () => { clearTimeout(failsafe); io.disconnect(); };
  }, [once]);

  const units = per === 'word' ? source.split(/(\s+)/) : Array.from(source);
  const hidden = {
    fade: { opacity: 0 },
    'fade-up': { opacity: 0, transform: 'translateY(0.5em)' },
    blur: { opacity: 0, filter: 'blur(8px)', transform: 'translateY(0.28em)' },
  }[preset] || { opacity: 0 };

  return (
    <Tag ref={ref} style={{ display: 'inline-block', ...style }} {...rest}>
      <span aria-hidden="true" style={{ display: 'inline' }}>
        {units.map((u, i) => (
          <span
            key={i}
            style={{
              display: 'inline-block', whiteSpace: 'pre', willChange: 'opacity, transform, filter',
              transition: reduced() ? 'none' : 'opacity 620ms cubic-bezier(.22,1,.36,1), transform 620ms cubic-bezier(.22,1,.36,1), filter 620ms cubic-bezier(.22,1,.36,1)',
              transitionDelay: `${delayN + i * step}s`,
              ...(on ? { opacity: 1, filter: 'blur(0px)', transform: 'none' } : hidden),
            }}
          >{u}</span>
        ))}
      </span>
      <span style={srOnly}>{source}</span>
    </Tag>
  );
}

const srOnly = { position: 'absolute', width: 1, height: 1, overflow: 'hidden', clip: 'rect(0 0 0 0)', whiteSpace: 'nowrap' };
function reduced() {
  return typeof window !== 'undefined' && window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}
function flatten(node) {
  if (node == null || node === false) return '';
  if (typeof node === 'string' || typeof node === 'number') return String(node);
  if (Array.isArray(node)) return node.map(flatten).join('');
  if (node.props) return flatten(node.props.children);
  return '';
}
