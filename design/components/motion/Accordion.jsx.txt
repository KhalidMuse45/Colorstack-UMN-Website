import React from 'react';

/**
 * Accordion. Each direct child is one item and should contain an element
 * marked `data-acc-title` and an element marked `data-acc-body`.
 */
export function Accordion({ allowMultiple = false, defaultOpen = -1, children, style, ...rest }) {
  const items = React.Children.toArray(children).filter(React.isValidElement);
  const [open, setOpen] = React.useState(() => (defaultOpen >= 0 ? [defaultOpen] : []));
  const toggle = i => setOpen(prev => (
    prev.includes(i) ? prev.filter(x => x !== i) : allowMultiple ? [...prev, i] : [i]
  ));
  return (
    <div style={{ display: 'flex', flexDirection: 'column', ...style }} {...rest}>
      {items.map((item, i) => (
        <Row key={i} node={item} isOpen={open.includes(i)} onToggle={() => toggle(i)} />
      ))}
    </div>
  );
}

function Row({ node, isOpen, onToggle }) {
  const bodyRef = React.useRef(null);
  const [h, setH] = React.useState(0);
  const parts = React.Children.toArray(node.props && node.props.children).filter(React.isValidElement);
  const title = parts.find(c => c.props && c.props['data-acc-title'] !== undefined) || parts[0];
  const body = parts.find(c => c.props && c.props['data-acc-body'] !== undefined) || parts[1];

  React.useEffect(() => {
    const el = bodyRef.current;
    if (!el) return;
    const measure = () => setH(el.scrollHeight);
    measure();
    const ro = typeof ResizeObserver !== 'undefined' ? new ResizeObserver(measure) : null;
    if (ro) ro.observe(el);
    return () => ro && ro.disconnect();
  }, [body]);

  return (
    <div style={{ borderBottom: '1px solid var(--line)' }}>
      <button
        type="button"
        onClick={onToggle}
        aria-expanded={isOpen}
        style={{
          width: '100%', display: 'flex', alignItems: 'center', gap: 14, textAlign: 'left',
          background: 'transparent', border: 'none', cursor: 'pointer', padding: '18px 2px', minHeight: 44,
          font: 'inherit', color: 'inherit',
        }}
      >
        <span
          aria-hidden="true"
          style={{
            flex: 'none', width: 22, height: 22, display: 'grid', placeItems: 'center',
            color: 'var(--maroon-light)', fontSize: 15, lineHeight: 1,
            transform: isOpen ? 'rotate(90deg)' : 'none',
            transition: 'transform 320ms cubic-bezier(.34,1.4,.5,1)',
          }}
        >›</span>
        <span style={{ flex: 1, minWidth: 0 }}>{title}</span>
      </button>
      <div
        style={{
          overflow: 'hidden',
          height: isOpen ? h : 0,
          opacity: isOpen ? 1 : 0,
          transition: 'height 420ms cubic-bezier(.22,1,.36,1), opacity 320ms ease-out',
        }}
      >
        <div ref={bodyRef} style={{ padding: '0 2px 20px 36px' }}>{body}</div>
      </div>
    </div>
  );
}
